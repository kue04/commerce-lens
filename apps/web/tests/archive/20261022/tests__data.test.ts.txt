test("支付 fixture 符合共享契约，并与人工预期及后端独立 SQL 样例一致", () => {
  const r = makeReport(
    "昨天华东支付金额为什么下降？",
    "test-session",
    "test-run",
  );
  assert.equal(Value.Check(CatalogSchema, catalog), true);
  assert.equal(Value.Check(ReportSchema, r), true);
  assert.deepEqual(r.comparison, {
    current: 7_000_000,
    baseline: 10_000_000,
    delta: -3_000_000,
    changeRate: -0.3,
  });
  assert.deepEqual(r.comparison, flow.report.comparison);
  assert.deepEqual(
    r.contributors.map((c) => c.delta),
    [-4_000_000, 1_000_000],
  );
  assert.equal(stockHours(r.evidences[2]!), 10);
  assert.equal(stockHours(r.evidences[2]!, true), 0);
});

test("库存池聚合保留分母，不反推逐小时库存；缺失不补零", () => {
  const inventory = structuredClone(makeReport("支付分析", "s", "r").evidences[2]!);
  inventory.signal = {
    type: "inventory",
    current: { expectedHours: 48, observedHours: 48, zeroHours: 10, coverage: 1 },

  inventory.signal.current.coverage = 0.5;
  assert.equal(stockHours(inventory), null);
  inventory.status = "missing";
  assert.equal(stockHours(inventory), null);
});

test("商品与渠道共同筛选，报告、证据及分组数值一致", () => {
  const query = {
    ...baseQuery,
    filters: { regionId: ["east"], skuId: ["nuts"], channelId: ["mall"] },
  };
  const snapshot = structuredClone(query);
  const r = makeReport("查看坚果商城表现", "session", "run", "success", query);
  assert.equal(Value.Check(ReportSchema, r), true);
  assert.deepEqual(r.comparison, {
    current: 1_500_000,
    baseline: 4_500_000,
    delta: -3_000_000,
    changeRate: -2 / 3,
  });
  assert.equal(r.contributors[0]?.delta, -3_000_000);
  assert.deepEqual(
    r.evidences[0]?.rows[0]?.slice(0, 3),
    [1_500_000, 4_500_000, -3_000_000],
  );
  assert.deepEqual(query, snapshot);
  const mall = makeReport("品牌商城表现", "s", "r", "success", {
    ...baseQuery,
    filters: { regionId: ["east"], channelId: ["mall"] },
  });
  assert.equal(mall.comparison.current, 4_500_000);
  assert.equal(mall.comparison.baseline, 7_000_000);
});

for (const cancelFirst of [true, false]) {
  test(`取消/完成竞态：${cancelFirst ? "最后阶段取消先到" : "完成先到，随后取消"}，终态不反转`, async (t) => {
    t.mock.timers.enable({ apis: ["setTimeout"] });
    const api = createApi("mock");
    const run = await api.start(`race-${cancelFirst}`, {
      question: "昨天华东支付金额为什么下降？", scenario: "success",
    });
    const terminal: string[] = [];
    let ready = false;
    const stop = api.subscribe(run.id, 0, (event) => {
      if (cancelFirst && event.type === "phase.changed" && event.payload.phaseName === "inspecting")
        void api.cancel(run.id);
      if (event.type === "report.ready") ready = true;
      if (event.type === "run.finished") terminal.push(event.payload.status);
    }, () => undefined, () => undefined);
    try {
      for (let i = 0; i < 5; i++) {
        t.mock.timers.tick(380);
        await Promise.resolve();
      }
      const result = await api.cancel(run.id);
      const expected = cancelFirst ? "cancelled" : "completed";
      assert.equal(result.status, expected);
      assert.equal((await api.run(run.id)).status, expected);
      assert.equal(Boolean(result.reportId), !cancelFirst);
      assert.equal(ready, !cancelFirst);
      assert.deepEqual(terminal, [expected]);
    } finally { stop(); }
  });
}
