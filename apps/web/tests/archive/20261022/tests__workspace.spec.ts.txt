  ["error", "这次分析未完成"],

test("刷新后的中断任务可重新分析，旧任务不会覆盖新报告", async ({ page }) => {
  await scenario(page, "hold");
  await send(page);
  await expect(page.locator(".running-state")).toBeVisible();
  const oldRunId = new URL(page.url()).searchParams.get("run");
  await page.reload();
  await expect(page.getByRole("heading", { name: "模拟分析已中断" })).toBeVisible();
  await page.getByRole("button", { name: "重新分析", exact: true }).click();
  await expect(page.getByRole("heading", { name: "模拟分析已中断" })).toHaveCount(0);
  await expect(page.getByRole("heading", { name: "华东销售，哪里变了？" })).toBeVisible();
  const savedRuns = await page.evaluate(() => JSON.parse(localStorage.getItem("commerce-lens:demo-runs:v1")!));
  const interrupted = savedRuns.find((r: { id: string }) => r.id === oldRunId);
  const completed = savedRuns.find((r: { parentRunId?: string }) => r.parentRunId === oldRunId);
  expect(interrupted.status).toBe("interrupted");
  expect(completed.status).toBe("completed");
  expect(completed.id).not.toBe(oldRunId);
});

test("HTTP 契约样例能展示两条证据，未伪造库存与 Agent", async ({ page }) => {
  await page.addInitScript(() => {
    localStorage.setItem("commerce-lens:mode", "http");
  });
  await page.route("**/api/**", async (route) => {
    const url = new URL(route.request().url());
    if (url.pathname === "/api/catalog")
      return route.fulfill({ json: flow.catalog });
    if (url.pathname === "/api/sessions")
      return route.fulfill({
        json:
          route.request().method() === "POST"
            ? { sessionId: "example-session", datasetVersion: "acceptance-v1" }
            : [],
      });
    if (url.pathname.endsWith("/runs"))
      return route.fulfill({ status: 202, json: flow.startRun });
    if (url.pathname.endsWith("/events"))
      return route.fulfill({
        contentType: "text/event-stream",
        body: [...flow.events, flow.events[2]!]
          .map((e) => `id: ${e.seq}\ndata: ${JSON.stringify(e)}\n\n`)
          .join(""),
      });
    if (url.pathname === "/api/runs/example-run")
      return route.fulfill({ json: flow.run });
    if (url.pathname.startsWith("/api/reports/"))
      return route.fulfill({ json: flow.report });
    if (url.pathname.startsWith("/api/evidences/"))
      return route.fulfill({ json: flow.evidence });
    return route.fulfill({ status: 404, json: flow.error });
  });
  await page.goto("/");
  await send(page);
  await expect(
    page.getByRole("heading", { name: "支付金额变化分析" }),
  ).toBeVisible();
  await expect(page.locator(".hero-number")).toContainText("70,000");
  await expect(page.getByRole("button", { name: "查看证据 E03" })).toHaveCount(
    0,
  );
  await page.getByRole("button", { name: "查看分析口径" }).click();
  await expect(page.locator(".evidence-table")).toContainText("¥70,000.00");
});

for (const status of ["cancelled", "completed"] as const) {
  test(`report.ready 后等待权威终态，取消请求返回 ${status}`, async ({ page }) => {
    await page.addInitScript(() => localStorage.setItem("commerce-lens:mode", "http"));
    await page.route("**/api/**", async (route) => {
      const path = new URL(route.request().url()).pathname;
      if (path === "/api/catalog") return route.fulfill({ json: flow.catalog });
      if (path === "/api/sessions") return route.fulfill({ json: route.request().method() === "POST"
        ? { sessionId: "example-session", datasetVersion: "acceptance-v1" } : [] });
      if (path.endsWith("/runs")) return route.fulfill({ status: 202, json: flow.startRun });
      if (path.endsWith("/events")) return route.fulfill({ contentType: "text/event-stream",
        body: flow.events.filter((e) => e.type !== "run.finished")
          .map((e) => `id: ${e.seq}\ndata: ${JSON.stringify(e)}\n\n`).join("") });
      if (path.endsWith("/cancel")) return route.fulfill({ json: { ...flow.run, status,
        ...(status === "cancelled" ? { reportId: undefined } : {}) } });
      if (path === "/api/runs/example-run") return route.fulfill({ json: { ...flow.run, status: "running" } });
      if (path.startsWith("/api/reports/")) return route.fulfill({ json: flow.report });
      return route.fulfill({ status: 404, json: flow.error });
    });
    await page.goto("/");
    await expect(page.getByRole("textbox", { name: "输入分析问题" })).toBeEnabled();
    await send(page);
    await expect(page.getByText("连接中断，正在从最后事件序号恢复。任务未取消。", { exact: true })).toBeVisible();
    await expect(page.locator(".report")).toHaveCount(0);
    await page.getByRole("button", { name: "取消本次分析", exact: true }).click();
    if (status === "cancelled") {
      await expect(page.getByRole("heading", { name: "分析已取消" })).toBeVisible();
      await expect(page.locator(".report")).toHaveCount(0);
    } else {
      await expect(page.getByRole("heading", { name: "支付金额变化分析" })).toBeVisible();
      await expect(page.locator(".hero-number")).toContainText("70,000");
    }
    await expect(page.getByRole("textbox", { name: "输入分析问题" })).toBeEnabled();
    await expect(page.getByText("连接中断，正在从最后事件序号恢复。任务未取消。", { exact: true })).toHaveCount(0);
  });
}

test("本地模拟页面不依赖外部网络资源", async ({ page }) => {
  const externalRequests: string[] = [];
  await page.route("**/*", (route) => {
    const origin = new URL(route.request().url()).origin;
    if (origin !== "http://127.0.0.1:5173") {
      externalRequests.push(route.request().url());
      return route.abort();
    }
    return route.continue();
  });
  await page.goto("/");
  await expect(
    page.getByRole("heading", { name: "华东销售，哪里变了？" }),
  ).toBeVisible();
  await page.evaluate(() => document.fonts.ready);
  expect(externalRequests).toEqual([]);
  expect(
    await page.evaluate(() => document.fonts.check('500 16px "DM Sans"')),
  ).toBe(true);
});
